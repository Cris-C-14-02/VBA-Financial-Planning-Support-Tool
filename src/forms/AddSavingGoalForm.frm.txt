Option Explicit

Private Sub SubmitBtn_Click()
    Dim begins As Date, ends As Date, opening As Currency, target As Currency
    On Error GoTo Failed
    If Not TryDateParts(txtBeginningYear.value, txtBeginningMonth.value, txtBeginningDay.value, begins) Then MsgBox "Enter a valid start date.", vbExclamation: Exit Sub
    If Not TryDateParts(txtExpectedYear.value, txtExpectedMonth.value, txtExpectedDay.value, ends) Then MsgBox "Enter a valid end date.", vbExclamation: Exit Sub
    If ends < begins Then MsgBox "End date must be on or after start date.", vbExclamation: Exit Sub
    If Not TryMoney(txtCurrentSaving.value, opening, True) Then MsgBox "Enter a nonnegative opening balance.", vbExclamation: Exit Sub
    If Not TryMoney(txtSavingGoal.value, target) Then MsgBox "Enter a positive savings target.", vbExclamation: Exit Sub
    SaveGoal begins, ends, opening, target
    ClearFinancialReport
    MsgBox "Goal saved. The latest goal is used for the report.", vbInformation
    Me.Hide
    Exit Sub
Failed:
    MsgBox Err.description, vbExclamation
End Sub
